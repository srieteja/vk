   ```go
   func (s *DBStore) DeleteByToken(ctx context.Context, token string) error {
       return s.db.WithContext(ctx).
           Where("token = ?", token).
           Delete(&models.Session{}).Error
   }

   func (s *DBStore) DeleteByUserID(ctx context.Context, userID uint, userType string) error {
       return s.db.WithContext(ctx).
           Where("user_id = ? AND user_type = ?", userID, userType).
           Delete(&models.Session{}).Error
   }

   func (s *DBStore) DeleteAllSessions(ctx context.Context) error {
       return s.db.WithContext(ctx).
           Where("1 = 1").
           Delete(&models.Session{}).Error
   }
   ```

3. **Implement in RedisStore** (Redis-backed sessions):
   ```go
   func (s *RedisStore) DeleteByToken(ctx context.Context, token string) error {
       key := fmt.Sprintf("session:%s", token)
       return s.client.Del(ctx, key).Err()
   }

   func (s *RedisStore) DeleteByUserID(ctx context.Context, userID uint, userType string) error {
       // Use SCAN to find all keys matching pattern
       pattern := fmt.Sprintf("session:user:%d:%s:*", userID, userType)
       var cursor uint64
       for {
           keys, nextCursor, err := s.client.Scan(ctx, cursor, pattern, 100).Result()
           if err != nil {
               return err
           }
           if len(keys) > 0 {
               if err := s.client.Del(ctx, keys...).Err(); err != nil {
                   return err
               }
           }
           cursor = nextCursor
           if cursor == 0 {
               break
           }
       }
       return nil
   }

   func (s *RedisStore) DeleteAllSessions(ctx context.Context) error {
       pattern := "session:*"
       var cursor uint64
       for {
           keys, nextCursor, err := s.client.Scan(ctx, cursor, pattern, 100).Result()
           if err != nil {
               return err
           }
           if len(keys) > 0 {
               if err := s.client.Del(ctx, keys...).Err(); err != nil {
                   return err
               }
           }
           cursor = nextCursor
           if cursor == 0 {
               break
           }
       }
       return nil
   }
   ```

4. **Implement in HybridStore** (calls both Redis and DB):
   ```go
   func (s *HybridStore) DeleteByToken(ctx context.Context, token string) error {
       // Delete from Redis (ignore error if not exists)
       _ = s.redisStore.DeleteByToken(ctx, token)
       // Delete from DB (return error)
       return s.dbStore.DeleteByToken(ctx, token)
   }

   func (s *HybridStore) DeleteByUserID(ctx context.Context, userID uint, userType string) error {
       _ = s.redisStore.DeleteByUserID(ctx, userID, userType)
       return s.dbStore.DeleteByUserID(ctx, userID, userType)
   }

   func (s *HybridStore) DeleteAllSessions(ctx context.Context) error {
       _ = s.redisStore.DeleteAllSessions(ctx)
       return s.dbStore.DeleteAllSessions(ctx)
   }
   ```
